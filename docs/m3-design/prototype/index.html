// ================= SMARTQUEUE - FULL SYSTEM ================= // Roles: CUSTOMER, TELLER, MANAGER, IT // Services: Deposit, Withdrawals, Customer Service // ---------- 1. DATA ---------- const SERVICES = { DEPOSIT: { id: 'DEPOSIT', name: 'Deposit', prefix: 'D', color: '#2E7D32' }, WITHDRAWAL: { id: 'WITHDRAWAL', name: 'Withdrawals', prefix: 'W', color: '#E65100' }, CUSTOMER_SERVICE: { id: 'CUSTOMER_SERVICE', name: 'Customer Service', prefix: 'C', color: '#1565C0' } }; const ROLES = ['CUSTOMER', 'TELLER', 'MANAGER', 'IT']; // Load from localStorage or start fresh let DB = JSON.parse(localStorage.getItem('smartqueue_db')) || { tickets: [], users: [ { id: 'U1', name: 'Admin IT', role: 'IT', username: 'admin', password: 'admin123', status: 'Active' }, { id: 'U2', name: 'Jean Teller', role: 'TELLER', username: 'jean', password: '1234', counter: 1, status: 'Active' }, { id: 'U3', name: 'Mary Manager', role: 'MANAGER', username: 'mary', password: '1234', status: 'Active' } ], counters: [ { id: 1, name: 'Counter 1', service: 'DEPOSIT', teller: 'Jean Teller', status: 'Open' }, { id: 2, name: 'Counter 2', service: 'WITHDRAWAL', teller: '-', status: 'Open' }, { id: 3, name: 'Counter 3', service: 'CUSTOMER_SERVICE', teller: '-', status: 'Closed' }, { id: 4, name: 'Counter 4', service: 'DEPOSIT', teller: '-', status: 'Open' } ], countersTicket: { DEPOSIT: 0, WITHDRAWAL: 0, CUSTOMER_SERVICE: 0 }, currentUser: null, logs: [] }; function save() { localStorage.setItem('smartqueue_db', JSON.stringify(DB)); log(`DB saved at ${new Date().toLocaleTimeString()}`); } function log(msg) { DB.logs.unshift({ time: new Date().toISOString(), msg }); if (DB.logs.length > 100) DB.logs.pop(); } // ---------- 2. AUTH ---------- function login(username, password) { const user = DB.users.find(u => u.username === username && u.password === password && u.status === 'Active'); if (!user) return { success: false, message: 'Wrong username or password' }; DB.currentUser = user; save(); return { success: true, user, redirect: getDashboardForRole(user.role) }; } function logout() { DB.currentUser = null; save(); window.location.href = 'login.html'; } function getDashboardForRole(role) { if (role === 'TELLER') return 'teller.html'; if (role === 'MANAGER') return 'manager.html'; if (role === 'IT') return 'it.html'; return 'customer.html'; } function requireRole(...allowed) { if (!DB.currentUser || !allowed.includes(DB.currentUser.role)) { window.location.href = 'login.html'; return false; } return true; } // ---------- 3. CUSTOMER - GET TICKET ---------- function getTicket(serviceId, customerName = 'Guest', phone = '') { if (!SERVICES[serviceId]) return { success: false, message: 'Invalid service' }; DB.countersTicket[serviceId]++; const num = String(DB.countersTicket[serviceId]).padStart(3, '0'); const ticketNo = `${SERVICES[serviceId].prefix}-${num}`; const waitingBefore = DB.tickets.filter(t => t.service === serviceId && t.status === 'WAITING').length; const waitMin = waitingBefore * 4; // 4 min per person const ticket = { id: Date.now(), number: ticketNo, service: serviceId, serviceName: SERVICES[serviceId].name, customerName, phone, status: 'WAITING', // WAITING | SERVING | SERVED | SKIPPED position: waitingBefore + 1, waitMin, createdAt: new Date().toISOString(), counter: null, teller: null }; DB.tickets.push(ticket); save(); return { success: true, ticket, message: `Your ticket is ${ticketNo}. Position ${ticket.position}, ~${waitMin} min` }; } function getMyTicket(ticketId) { return DB.tickets.find(t => t.id === ticketId); } function getWaitingList(serviceId = null) { let list = DB.tickets.filter(t => t.status === 'WAITING' || t.status === 'SERVING'); if (serviceId) list = list.filter(t => t.service === serviceId); return list.sort((a, b) => a.createdAt.localeCompare(b.createdAt)); } function getNowServing(serviceId = null) { let list = DB.tickets.filter(t => t.status === 'SERVING'); if (serviceId) list = list.filter(t => t.service === serviceId); return list[0] || null; } // ---------- 4. TELLER ---------- function tellerCallNext(tellerName, counterId, serviceId) { const next = getWaitingList(serviceId)[0]; if (!next) return { success: false, message: 'No one waiting' }; // Mark previous SERVING as SERVED if any for this counter DB.tickets.forEach(t => { if (t.counter === counterId && t.status === 'SERVING') t.status = 'SERVED'; }); next.status = 'SERVING'; next.counter = counterId; next.teller = tellerName; next.calledAt = new Date().toISOString(); save(); return { success: true, ticket: next, message: `Now serving ${next.number}` }; } function tellerComplete(ticketId) { const t = DB.tickets.find(x => x.id === ticketId); if (!t) return { success: false, message: 'Ticket not found' }; t.status = 'SERVED'; t.servedAt = new Date().toISOString(); save(); return { success: true, ticket: t }; } function tellerSkip(ticketId) { const t = DB.tickets.find(x => x.id === ticketId); if (!t) return { success: false, message: 'Ticket not found' }; t.status = 'SKIPPED'; save(); return { success: true, ticket: t }; } function tellerCallAgain(ticketId) { const t = DB.tickets.find(x => x.id === ticketId); if (!t) return { success: false, message: 'Ticket not found' }; t.calledAt = new Date().toISOString(); save(); return { success: true, message: `Calling ${t.number} again` }; } // ---------- 5. MANAGER ---------- function getManagerStats() { const today = new Date().toDateString(); const todayTickets = DB.tickets.filter(t => new Date(t.createdAt).toDateString() === today); const served = todayTickets.filter(t => t.status === 'SERVED').length; const waiting = todayTickets.filter(t => t.status === 'WAITING').length; const serving = todayTickets.filter(t => t.status === 'SERVING').length; // Avg wait - simulated from served count const avgWait = served > 0 ? Math.round(todayTickets.reduce((a, t) => a + (t.waitMin || 0), 0) / todayTickets.length) : 0; const byService = {}; for (let key in SERVICES) { byService[key] = { name: SERVICES[key].name, waiting: todayTickets.filter(t => t.service === key && t.status === 'WAITING').length, served: todayTickets.filter(t => t.service === key && t.status === 'SERVED').length }; } const byTeller = {}; DB.tickets.filter(t => t.teller).forEach(t => { if (!byTeller[t.teller]) byTeller[t.teller] = { served: 0, skipped: 0 }; if (t.status === 'SERVED') byTeller[t.teller].served++; if (t.status === 'SKIPPED') byTeller[t.teller].skipped++; }); return { total: todayTickets.length, served, waiting, serving, avgWait, activeCounters: DB.counters.filter(c => c.status === 'Open').length, byService, byTeller }; } function getReports(fromDate, toDate) { return DB.tickets.filter(t => { const d = new Date(t.createdAt); return d >= new Date(fromDate) && d <= new Date(toDate); }); } // ---------- 6. IT ADMIN ---------- function itAddUser({ name, role, username, password, counter }) { if (DB.users.find(u => u.username === username)) return { success: false, message: 'Username already exists' }; if (!ROLES.includes(role)) return { success: false, message: 'Invalid role' }; const user = { id: 'U' + Date.now(), name, role, username, password, counter: counter || null, status: 'Active' }; DB.users.push(user); save(); return { success: true, user }; } function itEditUser(userId, updates) { const u = DB.users.find(x => x.id === userId); if (!u) return { success: false, message: 'User not found' }; Object.assign(u, updates); save(); return { success: true, user: u }; } function itDeleteUser(userId) { DB.users = DB.users.filter(u => u.id !== userId); save(); return { success: true }; } function itToggleCounter(counterId) { const c = DB.counters.find(x => x.id === counterId); if (!c) return { success: false, message: 'Counter not found' }; c.status = c.status === 'Open' ? 'Closed' : 'Open'; save(); return { success: true, counter: c }; } function itResetQueue() { DB.tickets = []; DB.countersTicket = { DEPOSIT: 0, WITHDRAWAL: 0, CUSTOMER_SERVICE: 0 }; save(); return { success: true, message: 'Queue reset' }; } function getSystemHealth() { return { status: 'Online', uptime: '99.9%', totalTickets: DB.tickets.length, totalUsers: DB.users.length, activeCounters: DB.counters.filter(c => c.status === 'Open').length, logs: DB.logs.slice(0, 20) }; } // ---------- 7. UI HELPERS ---------- function renderQueueBoard(containerId, serviceId = null) { const el = document.getElementById(containerId); if (!el) return; const list = getWaitingList(serviceId); const now = getNowServing(serviceId); el.innerHTML = `
Now Serving: ${now ? now.number : '---'} - ${now ? now.serviceName : ''}
Waiting: ${list.length}
    ${list.map((t, i) => `
  • ${i + 1}. ${t.number} - ${t.serviceName} - ${t.customerName} - ~${t.waitMin} min
    • `).join('')}
`; } function renderManagerDashboard() { const s = getManagerStats(); const el = document.getElementById('managerStats'); if (el) el.innerHTML = `

Total Today: ${s.total} | Served: ${s.served} | Waiting: ${s.waiting} | Avg Wait: ${s.avgWait} min | Active Counters: ${s.activeCounters}

`; } // Auto-refresh every 5 seconds for live boards setInterval(() => { if (document.getElementById('queueBoard')) renderQueueBoard('queueBoard'); if (document.getElementById('managerStats')) renderManagerDashboard(); }, 5000); // ---------- 8.
